   579→    isSaving: Boolean,
   580→    saveLabel: String = "ذخیره پیش‌نویس",
   581→    finalizeLabel: String = "نهایی‌سازی",
   582→    onSaveDraft: () -> Unit,

   593→            },
   594→            confirmButton = {
   595→                Button(
   596→                    onClick = {
   597→                        showFinalizeDialog = false
   598→                        onFinalize()
   599→                    }
   600→                ) { Text("تأیید و نهایی‌سازی") }
   601→            },
   602→            dismissButton = {

   606→            }
   607→        )
   608→    }

   610→    Row(

   612→            .fillMaxWidth()

   640→private fun Int.toPersian(): String {
   641→    val persianDigits = arrayOf('۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹')
   642→    return this.toString().map { c ->
   643→        if (c.isDigit()) persianDigits[c - '0'] else c
   644→    }.joinToString("")
   645→}
